    # ========================================================

    def create_panel(self, parent):

        border = tk.Frame(

            parent,

            bg=self.CYAN

        )


        panel = tk.Frame(

            border,

            bg=self.PANEL

        )


        panel.pack(

            fill="both",

            expand=True,

            padx=1,

            pady=1

        )


        return border, panel


    # ========================================================

    # ========================================================

    def create_button(
        self,
        parent,
        text,
        command,
        icon="◈",
        color=None

        if color is None:

            color = self.CYAN


        button = tk.Button(

            parent,

            text=f"{icon}  {text}",

            command=command,

            font=(

                "Consolas",

                10,

                "bold"

            bg=self.PANEL_2,

            activebackground="#102A30",

            activeforeground="#FFFFFF",

            bd=0,

            padx=15,

            pady=15,

            cursor="hand2",

            anchor="w",

            highlightthickness=1,

            highlightbackground=self.CYAN_DARK

        return button

    # ========================================================
    # SISTEMA DE LOG
    # ========================================================

    def log(self, message):

        def write_log():

            timestamp = datetime.now().strftime(
                "%H:%M:%S"
            )

            self.status.insert(

                tk.END,

                f"[{timestamp}] {message}\n"

            )

            self.status.see(
                tk.END
            )


        self.root.after(
            0,
            write_log
        )


    # ========================================================
    # SELECCIONAR VIDEO
    # ========================================================

    def select_video(self):

        video = filedialog.askopenfilename(

            title="Seleccionar evidencia de video",

            filetypes=[

                (

                    "Archivos de video",

                    "*.mp4 *.avi *.mov *.mkv *.wmv *.mpeg *.webm"

                ),

                (

                    "Todos los archivos",

                    "*.*"

                )

            ]

        )


        if video:

            self.video_path = video


            self.case_dir = create_case_directory(
                video
            )


            filename = os.path.basename(
                video
            )


            self.video_label.config(

                text=(

                    f"ARCHIVO:\n"
                    f"{filename}\n\n"

                    f"RUTA:\n"
                    f"{video}\n\n"

                    f"CASO:\n"
                    f"{self.case_dir}"

                ),

                fg=self.TEXT

            )


            self.case_status.config(

                text=(

                    "● EVIDENCIA CARGADA\n\n"

                    "Integridad pendiente\n"
                    "de validación."

                ),

                fg=self.SUCCESS

            )


            self.log(
                "[+] Evidencia digital seleccionada."
            )

            self.log(
                f"[+] Archivo: {filename}"
            )

            self.log(
                f"[+] Caso creado: {self.case_dir}"
            )


    # ========================================================
    # VALIDAR VIDEO
    # ========================================================

    def check_video(self):

        if not self.video_path:

            messagebox.showwarning(

                "Advertencia",

                "Primero seleccione un video."

            )

            return False


        return True


    # ========================================================
    # EJECUTAR THREAD
    # ========================================================

    def run_thread(self, function):

        thread = threading.Thread(

            target=function,

            daemon=True

        )

        thread.start()


    # ========================================================
    # HASH
    # ========================================================

    def hash_analysis(self):

        if not self.check_video():
            return

        self.run_thread(
            self._hash_analysis
        )


    def _hash_analysis(self):

            "[ANALYSIS] Generando hashes..."

            "[SUCCESS] Hashes generados."

    def _full_analysis(self):

        try:

            self.log(
                "[SYSTEM] =================================="
            )

            self.log(
                "[SYSTEM] INICIANDO ANALISIS FORENSE"
            )

            self.log(
                "[SYSTEM] =================================="

                self.video_path,

                self.case_dir

            )

                "[SUCCESS] ANALISIS COMPLETADO"
            )

            self.log(
                f"[REPORT] {report}"

            def update_interface():

                self.case_status.config(

                    text=(

                        "● ANÁLISIS COMPLETADO\n\n"

                    "Análisis terminado",

                0,
                update_interface
            )


        except Exception as e:

            self.log(
                f"[ERROR] {str(e)}"
            )


            self.root.after(

                0,

                lambda:

                messagebox.showerror(

                    "Error",

                    str(e)

                )

            )
