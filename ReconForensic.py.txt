def safe_filename(text):
    """Genera un nombre seguro para archivos."""
    return "".join(
        c if c.isalnum() or c in ("_", "-", ".") else "_"
        for c in text
    )


def create_case_directory(video_path):
    """
    Crea un directorio independiente para cada analisis.
    """

    video_name = os.path.basename(video_path)
    name = os.path.splitext(video_name)[0]

    timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")

    case_name = f"{safe_filename(name)}_{timestamp}"

    case_dir = os.path.join(
        OUTPUT_DIR,
        case_name
    )

    os.makedirs(case_dir, exist_ok=True)

    return case_dir


def run_command(command):
    """
    Ejecuta comandos del sistema y retorna salida.
    """

    try:

        result = subprocess.run(
            command,
            stdout=subprocess.PIPE,
            stderr=subprocess.PIPE,
            text=True
        )

        return {
            "returncode": result.returncode,
            "stdout": result.stdout,
            "stderr": result.stderr
        }

    except Exception as e:

        return {
            "returncode": -1,
            "stdout": "",
            "stderr": str(e)
        }


# ============================================================
# ANALISIS DE HASH
# ============================================================

def generate_hash(video_path, output_dir):

    algorithms = [
        "md5",
        "sha1",
        "sha256",
        "sha512"
    ]

    results = {}

    for algorithm in algorithms:

        hasher = hashlib.new(algorithm)

        with open(video_path, "rb") as file:

            while True:

                data = file.read(1024 * 1024)

                if not data:
                    break

                hasher.update(data)

        results[algorithm.upper()] = hasher.hexdigest()

    output_file = os.path.join(
        output_dir,
        "hashes.txt"
    )

    with open(output_file, "w") as file:

        file.write("VIDEO FORENSIC HASH REPORT\n")
        file.write("=" * 60 + "\n\n")

        file.write(
            f"Archivo: {video_path}\n"
        )

        file.write(
            f"Fecha: {datetime.now()}\n\n"
        )

        for algorithm, value in results.items():

            file.write(
                f"{algorithm}: {value}\n"
            )

    return results, output_file

# ============================================================

def extract_metadata(video_path, output_dir):

    output_file = os.path.join(
        output_dir,
        "metadata_exiftool.txt"
    )

    command = [
        "exiftool",
        "-G1",
        "-a",
        "-s",
        video_path
    ]

    result = run_command(command)

    with open(
        output_file,
        "w",
        encoding="utf-8"
    ) as file:

        file.write(result["stdout"])

        if result["stderr"]:

            file.write("\n\nERRORES:\n")

            file.write(
                result["stderr"]
            )

    return output_file

    output_file = os.path.join(

    command = [

        "ffprobe",

        "-v",
        "quiet",

        "-print_format",

        "-show_frames",

        video_path

    result = run_command(command)

    with open(
        output_file,

        encoding="utf-8"

        file.write(result["stdout"])

    return output_file


# ============================================================
# OBTENER INFORMACION DEL VIDEO
# ============================================================

def get_video_info(video_path):

    capture = cv2.VideoCapture(video_path)

    if not capture.isOpened():

        raise RuntimeError(
            "No fue posible abrir el video."
        )

    fps = capture.get(
        cv2.CAP_PROP_FPS
    )

    frame_count = int(
        capture.get(
            cv2.CAP_PROP_FRAME_COUNT
        )
    )

    width = int(
        capture.get(
            cv2.CAP_PROP_FRAME_WIDTH
        )
    )

    height = int(
        capture.get(
            cv2.CAP_PROP_FRAME_HEIGHT
        )
    )

    duration = 0

    if fps > 0:

        duration = frame_count / fps

    capture.release()

    return {

        "fps": fps,

        "frames": frame_count,

        "width": width,

        "height": height,

        "duration": duration
    }

def extract_frames(video_path, output_dir):

    frames_dir = os.path.join(
        output_dir,
        "frames"
    )

    os.makedirs(
        frames_dir,
        exist_ok=True
    )

    command = [

        "ffmpeg",

        "-i",
        video_path,

        "-vsync",
        "0",

        os.path.join(
            frames_dir,
            "frame_%08d.png"
        )
    ]

    result = run_command(command)

    return frames_dir


# ============================================================
# CALCULO DE DIFERENCIA ENTRE FRAMES
# ============================================================

def frame_difference(frame1, frame2):

    gray1 = cv2.cvtColor(
        frame1,
        cv2.COLOR_BGR2GRAY
    )

    gray2 = cv2.cvtColor(
        frame2,
        cv2.COLOR_BGR2GRAY
    )

    difference = cv2.absdiff(
        gray1,
        gray2
    )

    score = np.mean(
        difference
    )

    return score


# ============================================================
# DETECCION DE FRAMES DUPLICADOS
# ============================================================

def detect_duplicate_frames(
    video_path,
    output_dir,
    threshold=1.0
):

    capture = cv2.VideoCapture(video_path)

    duplicates = []

    previous_frame = None

    frame_number = 0

    while True:

        success, frame = capture.read()

        if not success:

            break

        if previous_frame is not None:

            score = frame_difference(
                previous_frame,
                frame
            )

            if score < threshold:

                duplicates.append({

                    "frame": frame_number,

                    "difference_score": float(score)

                })

        previous_frame = frame.copy()

        frame_number += 1

    capture.release()

    output_file = os.path.join(
        output_dir,
        "duplicate_frames.json"
    )

    with open(
        output_file,
        "w"
    ) as file:

        json.dump(
            duplicates,
            file,
            indent=4
        )

    return duplicates


# ============================================================
# DETECCION DE CAMBIOS BRUSCOS / CORTES
# ============================================================

def detect_cuts(
    video_path,
    output_dir,
    threshold=30
):

    capture = cv2.VideoCapture(video_path)

    cuts = []

    previous_frame = None

    frame_number = 0

    while True:

        success, frame = capture.read()

        if not success:

            break

        if previous_frame is not None:

            score = frame_difference(
                previous_frame,
                frame
            )

            if score > threshold:

                cuts.append({

                    "frame": frame_number,

                    "difference_score": float(score)

                })

        previous_frame = frame.copy()

        frame_number += 1

    capture.release()

    output_file = os.path.join(
        output_dir,
        "possible_cuts.json"
    )

    with open(
        output_file,
        "w"
    ) as file:

        json.dump(
            cuts,
            file,
            indent=4
        )

    return cuts


# ============================================================
# ANALISIS DE ILUMINACION
# ============================================================

def detect_lighting_changes(
    video_path,
    output_dir,
    threshold=25
):

    capture = cv2.VideoCapture(video_path)

    changes = []

    previous_brightness = None

    frame_number = 0

    while True:

        success, frame = capture.read()

        if not success:

            break

        gray = cv2.cvtColor(
            frame,
            cv2.COLOR_BGR2GRAY
        )

        brightness = np.mean(gray)

        if previous_brightness is not None:

            difference = abs(
                brightness -
                previous_brightness
            )

            if difference > threshold:

                changes.append({

                    "frame": frame_number,

                    "brightness": float(
                        brightness
                    ),

                    "difference": float(
                        difference
                    )

                })

        previous_brightness = brightness

        frame_number += 1

    capture.release()

    output_file = os.path.join(
        output_dir,
        "lighting_changes.json"
    )

    with open(
        output_file,
        "w"
    ) as file:

        json.dump(
            changes,
            file,
            indent=4
        )

    return changes


# ============================================================
# DETECCION DE INCONSISTENCIAS VISUALES
# ============================================================

def detect_visual_anomalies(
    video_path,
    output_dir
):

    capture = cv2.VideoCapture(video_path)

    anomalies = []

    previous_frame = None

    frame_number = 0

    while True:

        success, frame = capture.read()

        if not success:

            break

        gray = cv2.cvtColor(
            frame,
            cv2.COLOR_BGR2GRAY
        )

        blur_value = cv2.Laplacian(
            gray,
            cv2.CV_64F
        ).var()

        brightness = np.mean(
            gray
        )

        if previous_frame is not None:

            difference = frame_difference(
                previous_frame,
                frame
            )

            # Valores que pueden indicar
            # anomalías visuales importantes.

            if (
                blur_value < 20
                or brightness < 15
                or brightness > 245
            ):

                anomalies.append({

                    "frame": frame_number,

                    "blur_score": float(
                        blur_value
                    ),

                    "brightness": float(
                        brightness
                    ),

                    "frame_difference": float(
                        difference
                    )

                })

        previous_frame = frame.copy()

        frame_number += 1

    capture.release()

    output_file = os.path.join(
        output_dir,
        "visual_anomalies.json"
    )

    with open(
        output_file,
        "w"
    ) as file:

        json.dump(
            anomalies,
            file,
            indent=4
        )

    return anomalies


# ============================================================
# ANALISIS DE POSIBLES FRAMES ELIMINADOS O INSERTADOS
# ============================================================

def detect_temporal_anomalies(
    video_path,
    output_dir
):

    capture = cv2.VideoCapture(video_path)

    anomalies = []

    frames = []

    while True:

        success, frame = capture.read()

        if not success:

            break

        small = cv2.resize(
            frame,
            (160, 90)
        )

        gray = cv2.cvtColor(
            small,
            cv2.COLOR_BGR2GRAY
        )

        frames.append(
            gray
        )

    capture.release()

    for i in range(
        1,
        len(frames) - 1
    ):

        previous_frame = frames[i - 1]

        current_frame = frames[i]

        next_frame = frames[i + 1]

        diff_previous = np.mean(
            cv2.absdiff(
                previous_frame,
                current_frame
            )
        )

        diff_next = np.mean(
            cv2.absdiff(
                current_frame,
                next_frame
            )
        )

        bridge_difference = np.mean(
            cv2.absdiff(
                previous_frame,
                next_frame
            )
        )

        # Heurística básica:
        # un salto temporal fuerte entre
        # frame anterior y siguiente puede
        # indicar una discontinuidad.

        if bridge_difference > 40:

            anomalies.append({

                "frame": i,

                "previous_difference": float(
                    diff_previous
                ),

                "next_difference": float(
                    diff_next
                ),

                "temporal_jump": float(
                    bridge_difference
                ),

                "indicator":
                    "Possible temporal discontinuity"

            })

    output_file = os.path.join(
        output_dir,
        "temporal_anomalies.json"
    )

    with open(
        output_file,
        "w"
    ) as file:

        json.dump(
            anomalies,
            file,
            indent=4
        )

    return anomalies

def generate_report(
    video_path,
    output_dir
):

    report_file = os.path.join(
        output_dir,
        "forensic_report.txt"
    )

    info = get_video_info(
        video_path
    )

    hashes, hash_file = generate_hash(
        video_path,
        output_dir
    )

    metadata_file = extract_metadata(
        video_path,
        output_dir
    )

    duplicates = detect_duplicate_frames(
        video_path,
        output_dir
    )

    cuts = detect_cuts(
        video_path,
        output_dir
    )

    lighting = detect_lighting_changes(
        video_path,
        output_dir
    )

    temporal = detect_temporal_anomalies(
        video_path,
        output_dir
    )

    visual = detect_visual_anomalies(
        video_path,
        output_dir
    )

    with open(
        report_file,
        "w",
        encoding="utf-8"
    ) as file:

        file.write(
            "VIDEO FORENSIC ANALYSIS REPORT\n"
        )

        file.write(
            "=" * 70 + "\n\n"
        )

        file.write(
            f"Fecha del análisis: {datetime.now()}\n"
        )

        file.write(
            f"Archivo analizado: {video_path}\n\n"
        )

        file.write(
            "INFORMACION TECNICA\n"
        )

        file.write(
            "-" * 70 + "\n"
        )

        for key, value in info.items():

            file.write(
                f"{key}: {value}\n"
            )

        file.write("\n")

        file.write(
            "HASHES\n"
        )

        file.write(
            "-" * 70 + "\n"
        )

        for algorithm, value in hashes.items():

            file.write(
                f"{algorithm}: {value}\n"
            )

        file.write("\n")

        file.write(
            "RESULTADOS DE DETECCION\n"
        )

        file.write(
            "-" * 70 + "\n"
        )

        file.write(
            f"Frames duplicados detectados: "
            f"{len(duplicates)}\n"
        )

        file.write(
            f"Posibles cortes detectados: "
            f"{len(cuts)}\n"
        )

        file.write(
            f"Cambios de iluminación: "
            f"{len(lighting)}\n"
        )

        file.write(
            f"Posibles discontinuidades temporales: "
            f"{len(temporal)}\n"
        )

        file.write(
            f"Anomalías visuales: "
            f"{len(visual)}\n"
        )

        file.write("\n")

        file.write(
            "NOTA FORENSE\n"
        )

        file.write(
            "-" * 70 + "\n"
        )

        file.write(
            "Los resultados identificados por "
            "los algoritmos corresponden a "
            "indicadores técnicos y anomalías "
            "que requieren validación manual. "
            "Un cambio de compresión, movimiento "
            "rápido de cámara, transición legítima "
            "o recodificación puede producir "
            "resultados similares a una manipulación.\n"
        )

    return report_file
