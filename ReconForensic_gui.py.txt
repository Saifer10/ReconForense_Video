            f"{WINDOW_WIDTH}x{WINDOW_HEIGHT}"

        # TERMINAL
        # ----------------------------------------------------

        self.create_terminal()

            "[SYSTEM] HIDRA VIDEO FORENSICS LAB iniciado."

        self.log(
            "[READY] Esperando selección de evidencia digital."

    # ========================================================
    # CREAR BOTÓN PERSONALIZADO
    # ========================================================

    def create_forensic_button(
        self,
        parent,
        text,
        command
    ):

        button = tk.Button(

            parent,

            text=text,

            command=command,

            bg="#061719",

            fg=TEXT,

            activebackground=PANEL_HOVER,

            activeforeground=BORDER,

            font=FONT_BUTTON,

            relief="flat",

            bd=0,

            highlightbackground=BORDER_SOFT,

            highlightthickness=1,

            cursor="hand2",

            height=2
        )


        # Efecto visual al pasar el mouse

        button.bind(

            "<Enter>",

            lambda event:
            button.configure(
                bg=PANEL_HOVER,
                fg=BORDER
            )
        )


        button.bind(

            "<Leave>",

            lambda event:
            button.configure(
                bg="#061719",
                fg=TEXT
            )
        )


        return button

    def create_terminal(self):

        terminal_frame = tk.Frame(

            self.main_container,

            bg=PANEL_DARK,

            highlightbackground=BORDER,

            highlightthickness=1
        )

        terminal_frame.pack(

            fill="both",

        terminal_title = tk.Label(

            terminal_frame,

            text="[ ACTIVITY LOG / FORENSIC TERMINAL ]",

            fg=BORDER,

            bg=PANEL_DARK,

            font=FONT_SUBTITLE
        )

        terminal_title.pack(

            anchor="w",

            padx=15,

            pady=(10, 5)
        )


        self.terminal = scrolledtext.ScrolledText(

            terminal_frame,

            bg="#020B0C",

            fg=TEXT_DIM,

            insertbackground=BORDER,

            font=FONT_LOG,

            relief="flat",

            bd=0,

            height=9,

        self.terminal.pack(

        )


    # ========================================================
    # SELECCIONAR VIDEO
    # ========================================================

    def select_video(self):

        file_path = filedialog.askopenfilename(

            title="Seleccionar evidencia de video",

            filetypes=[

                (

                    "Archivos de video",

                    "*.mp4 *.avi *.mov *.mkv *.wmv *.webm *.flv"
                ),

                (

                    "Todos los archivos",

                    "*.*"
                )

            ]
        )


        if file_path:

            self.video_path = file_path


            file_name = os.path.basename(
                file_path
            )


            self.file_name_label.configure(

                text=file_name,

                fg=SUCCESS
            )


            self.case_status.configure(

                text="● EVIDENCIA CARGADA",

                fg=SUCCESS
            )


            self.log(

                f"[+] Video seleccionado: {file_name}"
            )


            self.log(

                "[READY] Evidencia preparada para análisis."
            )


    # ========================================================
    # FUNCIÓN TEMPORAL PARA LOS BOTONES
    # ========================================================

    def placeholder_action(self):

        self.log(

            "[INFO] Función visual preparada. "
            "Módulo forense pendiente de integración."
        )


    # ========================================================
    # LOG DEL TERMINAL
    # ========================================================

    def log(self, message):

        from datetime import datetime


        current_time = datetime.now().strftime(
            "%H:%M:%S"
        )


        log_message = (

            f"[{current_time}] "
            f"{message}\n"
        )


        self.terminal.insert(

            tk.END,

            log_message
        )


        self.terminal.see(
            tk.END
        )
